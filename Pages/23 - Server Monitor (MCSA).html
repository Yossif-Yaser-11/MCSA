Server Monitor - مراقبة السيرفر

شرح نظري وتطبيق عملي لإدارة Windows Server.

العودة إلى فهرس الدروس

مراقبة السيرفر ضرورية عشان تكتشف المشاكل قبل ما تأثر على المستخدمين. Windows Server بيوفّر أدوات مراقبة قوية.

Performance Monitor

بيراقب الأداء لحظة بلحظة:

# تشغيل Performance Monitor
perfmon

# من PowerShell
Get-Counter -Counter "\Processor(_Total)\% Processor Time" -Continuous -SampleInterval 5
Get-Counter -Counter "\Memory\Available MBytes"
Get-Counter -Counter "\LogicalDisk(C:)\% Disk Time"

Resource Monitor

# تشغيل Resource Monitor
resmon

# - Memory: الـ Working Set لكل Process

Event Viewer

# من PowerShell
# أحداث الـ System خلال آخر ساعة
Get-WinEvent -LogName System -MaxEvents 50 | Where-Object {$_.LevelDisplayName -eq "Error"} | Select TimeCreated, Id, Message

# أحداث الأمان
Get-WinEvent -LogName Security -FilterHashtable @{Id=4625; StartTime=(Get-Date).AddDays(-1)} | Select TimeCreated, Message

# Application Errors
Get-WinEvent -LogName Application | Where-Object {$_.LevelDisplayName -eq "Error"} | Group-Object -Property {$_.ProviderName} | Sort-Object Count -Descending | Select Count, Name

التطبيق العملى - Server Monitoring

تمرين 1: إعداد Performance Monitor

# إنشاء Data Collector Set
$name = "Daily_Performance"
$logpath = "C:\PerfLogs\Admin\Daily_Performance"
New-Item -Path $logpath -ItemType Directory -Force

# إضافة Counters
logman create counter $name -f csv -o "$logpath\log" -si 60 -c `
    "\Processor(_Total)\% Processor Time" `
    "\Memory\Available MBytes" `
    "\LogicalDisk(_Total)\% Disk Time" `
    "\Network Interface(*)\Bytes Total/sec"

# تشغيل التجميع
logman start $name

# إيقاف بعد ساعة
Start-Sleep -Seconds 3600
logman stop $name

تمرين 2: Alert على CPU مرتفع

# Script للتنبيه على CPU > 80%
while ($true) {
    $cpu = (Get-Counter "\Processor(_Total)\% Processor Time").CounterSamples[0].CookedValue
    
    if ($cpu -gt 80) {
        $message = "ALERT: CPU Usage = $([math]::Round($cpu, 1))% at $(Get-Date)"
        Write-EventLog -LogName Application -Source "Monitoring" -EventId 9001 -EntryType Warning -Message $message
        
        # إرسال Email
        Send-MailMessage -To "admin@networkat.local" -From "server@networkat.local" -Subject "CPU Alert" -Body $message -SmtpServer "mail.networkat.local"
        
        # الـ Top Processes
        Get-Process | Sort-Object CPU -Descending | Select -First 5 | ForEach-Object {
            Write-Host "$($_.Name): $([math]::Round($_.CPU, 1))s CPU time"
        }
    }
    Start-Sleep -Seconds 60
}

تمرين 3: Event Log Management

# مسح الـ Event Logs القديمة
Get-WinEvent -ListLog * | Where-Object {$_.RecordCount -gt 10000} | ForEach-Object {
    Write-Host "Clearing: $($_.LogName) ($($_.RecordCount) events)"
    wevtutil cl $_.LogName
}

# أرشفة الـ Logs
wevtutil epl Security "D:\Logs\Security_$(Get-Date -Format yyyyMMdd).evtx"
