Administrative Templates

الـ Administrative Templates (.admx files) بتضيف إعدادات إضافية للـ Group Policy. Microsoft وبرامج تانية زى Chrome و Firefox بتوفّر Templates خاصة بيها.

# نسخ الـ Templates للـ Central Store
# إنشاء Central Store
New-Item -Path "\\networkat.local\SYSVOL\networkat.local\Policies\PolicyDefinitions" -ItemType Directory

# نسخ الـ Templates
Copy-Item "C:\Windows\PolicyDefinitions\*" "\\networkat.local\SYSVOL\networkat.local\Policies\PolicyDefinitions\" -Recurse

AppLocker - تحكم فى البرامج

الـ AppLocker بيسمحلك تحدد بالظبط أنهى برامج يقدر المستخدم يشغّلها:

# إنشاء AppLocker Rules
$policy = Get-AppLockerPolicy -Local
# من الـ GUI:
# Computer Config → Windows Settings → Security Settings → Application Control → AppLocker
# Executable Rules: Allow المسارات المعتمدة
# Script Rules: منع Scripts من الـ Desktop
# Windows Installer Rules: تحديد الـ MSI المسموح بيها

# تفعيل خدمة AppLocker
Set-Service -Name AppIDSvc -StartupType Automatic
Start-Service AppIDSvc

Software Restriction Policies

# بديل أبسط لـ AppLocker
# Security Levels:
# Disallowed: منع كل البرامج إلا المستثنيات
# Unrestricted: السماح بكل البرامج إلا المحظورة

# Path Rule: السماح لمجلدات معينة فقط
# Hash Rule: السماح لملف معين بـ Hash محدد
# Certificate Rule: السماح للبرامج الموقعة بشهادة معينة

GPO Backup و Restore

# Backup GPO
Backup-GPO -Name "Security_Policy" -Path "D:\GPO_Backup"

# Backup كل الـ GPOs
Backup-GPO -All -Path "D:\GPO_Backup"

# Restore GPO
Restore-GPO -Name "Security_Policy" -Path "D:\GPO_Backup"

# Copy GPO لـ Domain تانى
Copy-GPO -SourceGpoName "Security_Policy" -SourceDomain "networkat.local" -TargetGpoName "Security_Policy" -TargetDomain "newdomain.local"

التطبيق العملى - Group Policy Advanced

تمرين 1: AppLocker

# تفعيل خدمة AppLocker
Set-Service -Name AppIDSvc -StartupType Automatic
Start-Service AppIDSvc

# إنشاء Rules من الـ GUI
# Computer Config → Security Settings → AppLocker
# Configure Rule Enforcement: Executable Rules → Enforced
# Create Default Rules (السماح لـ Windows وـ Program Files)
# إضافة Rule لمنع تشغيل من Desktop والـ Downloads

تمرين 2: Startup Scripts

# Script بيشتغل عند Startup
$startupScript = @"
# تحديث الوقت
w32tm /resync

# تنظيف الـ Temp Files
Remove-Item "$env:TEMP\*" -Force -Recurse -ErrorAction SilentlyContinue

# تسجيل الـ Event
Write-EventLog -LogName Application -Source "Company" -EventId 1001 -Message "Startup Script Completed"
"@

$startupScript | Out-File "\\networkat.local\SYSVOL\networkat.local\Policies\Scripts\startup.ps1"

# ربطه بـ GPO:
# Computer Config → Windows Settings → Scripts → Startup → PowerShell Scripts
